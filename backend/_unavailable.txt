function analyzeSymbol(session: SessionData, symbol: string): SymbolAnalysis[] {
  const points = session.prices.filter((p) => p.symbol === symbol).sort((a, b) => a.minute - b.minute);
  const first = points[0];
  const last = points.at(-1);
  if (!first || !last) return [];
  let min = first;
  let max = first;
  let biggest: SymbolAnalysis['biggestMove'] = null;
  const returns: number[] = [];
  for (let i = 0; i < points.length; i += 1) {
    const point = points[i]!;
    if (point.price < min.price) min = point;
    if (point.price > max.price) max = point;
    if (i === 0) continue;
    const previous = points[i - 1]!;
    if (previous.price <= 0) continue;
    const rate = (point.price - previous.price) / previous.price;
    returns.push(rate);
    if (!biggest || Math.abs(rate) > Math.abs(biggest.rate)) biggest = { minute: point.minute, rate: round(rate, 6) };
  }
  const mean = returns.reduce((s, r) => s + r, 0) / (returns.length || 1);
  const variance = returns.reduce((s, r) => s + (r - mean) ** 2, 0) / (returns.length || 1);
  return [
    {
      symbol,
      first: first.price,
      last: last.price,
      changeRate: first.price > 0 ? round((last.price - first.price) / first.price, 6) : 0,
      min: min.price,
      minMinute: min.minute,
      max: max.price,
      maxMinute: max.minute,
      volatility: round(Math.sqrt(variance), 6),
      biggestMove: biggest && biggest.rate !== 0 ? biggest : null,
    },
  ];
}

function analyzeEvent(session: SessionData, event: SessionData['events'][number]): EventAnalysis {
  const reacted = session.participants.filter((p) =>
    p.trades.some((t) => t.minute >= event.minute && t.minute <= event.minute + REACTION_WINDOW),
  ).length;
  return {
    id: event.id,
    type: event.type,
    minute: event.minute,
    label: sanitizeLabel(event.label),
    impacts: session.symbols.flatMap((symbol) => {
      const before = priceAt(session.prices, symbol, Math.max(0, event.minute - 1));
      const after = priceAt(session.prices, symbol, Math.min(session.durationMinutes, event.minute + IMPACT_WINDOW));
      if (before === undefined || after === undefined || before <= 0) return [];

    }),
    participants: session.participants.length,
    reacted,
  };
}

export function allowedFromData(data: unknown, durationMinutes = 60): AllowedNumbers {
  const any: number[] = [];
  const percent: number[] = [];
  const add = (n: number) => {
    const abs = Math.abs(n);
    any.push(abs);
    if (abs <= 1.0001) {
      any.push(abs * 100);
      percent.push(abs * 100);
    }
  };
  const walk = (value: unknown, key?: string) => {
    if (typeof value === 'number' && Number.isFinite(value)) add(value);
    else if (typeof value === 'string' && key === 'label') {
      const found = numbersIn(value).map((n) => Math.abs(n.value));
      any.push(...found);
      percent.push(...found);
    } else if (Array.isArray(value)) value.forEach((v) => walk(v));
    else if (value && typeof value === 'object') Object.entries(value).forEach(([k, v]) => walk(v, k));
  };
  walk(data);
  for (let i = 0; i <= Math.max(10, durationMinutes); i += 1) any.push(i);
  // Variations en points de pourcentage, Ã©cart de deux taux dÃ©jÃ  prÃ©sents : on les accepte sous forme de fractions.
  return { any, percent };
}

export async function evaluateParticipant(input: EvaluateInput): Promise<EvaluationResult> {

    throw new AiAccessError('Seul un enseignant peut gÃ©nÃ©rer un bilan.');
  }

  const metrics = computeMetrics(session, participantId);

  const request = buildEvaluatorRequest({ context, criteria: scores });

  const usage: LlmUsage = { inputTokens: 0, outputTokens: 0 };
  const messages: LlmMessage[] = [...request.messages];
  let parts: LlmReportParts | undefined;
  let model = 'none';
  let attempts = 0;

  for (let attempt = 1; attempt <= maxAttempts && !parts; attempt += 1) {
    attempts = attempt;
    try {
      const response = await provider.generate({ ...request, messages });
      model = response.model;
      usage.inputTokens += response.usage.inputTokens;
      usage.outputTokens += response.usage.outputTokens;
      const result = validateReportParts(response.text, context, scores, overall);
      if (result.ok) {
        parts = result.parts;
      } else {
        issues.push(...result.issues);
        messages.push({ role: 'assistant', content: response.text }, { role: 'user', content: buildRetryMessage(result.issues) });
      }
    } catch (error) {
      if (!(error instanceof LlmError)) throw error;
      issues.push({ code: 'llm_unavailable', message: error.message });
      break;
    }
  }

  const source: ReportTrace['source'] = parts ? 'llm' : 'rules';
  const finalParts = parts ?? buildFallbackParts(scores, overall);

    provider: provider.name,
    model,
    promptVersion: EVALUATOR_PROMPT_VERSION,
    rubricVersion: `${rubric.id}@${rubric.version}`,
    contextHash: hashValue(context),
    temperature: TEMPERATURE,
    attempts,
    usage,
    generatedAt: now().toISOString(),
    issues,
  };

  missingData: string[],

    const written = byId.get(score.id);
    return {
      ...score,
      comment: written?.comment ?? (score.status === 'evaluated' ? score.rule : (score.reason ?? score.rule)),
      commentEvidence: written?.evidence ?? [],
    };
  });
  return {
    alias: context.participant.alias,
    session: { id: context.session.id, run: context.session.run },
    rubric: { id: rubric.id, version: rubric.version, outOf: rubric.outOf },
    criteria,
    overall,
    summary: parts.summary,
    strengths: parts.strengths,
    improvements: parts.improvements,
    missingData,
  };

export function collectAllowedNumbers(context: AiContext, criteria: CriterionScore[], overall?: OverallScore): AllowedNumbers {
  const any: number[] = [];
  const percent: number[] = [];
  const add = (n: number) => {
    const abs = Math.abs(n);
    any.push(abs);
    if (abs <= 1.0001) {
      any.push(abs * 100); // fraction â†’ pourcentage
      percent.push(abs * 100);
    }
  };
  const walk = (value: unknown, key?: string) => {
    if (typeof value === 'number' && Number.isFinite(value)) add(value);
    else if (typeof value === 'string' && key === 'label') {
      const found = numbersIn(value).map((n) => Math.abs(n.value));
      any.push(...found);
      percent.push(...found);
    } else if (Array.isArray(value)) value.forEach((v) => walk(v));
    else if (isObject(value)) Object.entries(value).forEach(([k, v]) => walk(v, k));
  };
  walk(context);
  for (const c of criteria) {
    if (c.score !== null) any.push(c.score);
    any.push(c.outOf);
    if (c.value !== null) any.push(Math.abs(c.value));
    for (const text of [c.displayValue ?? '', c.rule]) {
      for (const n of numbersIn(text)) {
        any.push(Math.abs(n.value));
        if (/^[\s\u00a0\u202f]*%/.test(text.slice(n.end))) percent.push(Math.abs(n.value));
      }
    }
  }
  if (overall) {
    if (overall.score !== null) any.push(overall.score);
    any.push(overall.outOf, overall.evaluated, overall.total);
  }
  for (let i = 0; i <= Math.max(10, context.session.durationMinutes); i += 1) any.push(i); // minutes, petits comptes
  return { any, percent };
}

/** Interpolation linÃ©aire d'une sÃ©rie indexÃ©e en secondes de simulation. */
export function interpolate(points: DatasetPoint[], elapsed: number): number {
  if (points.length === 0) return 0;
  if (points.length === 1) return points[0].price;
  if (elapsed <= points[0].offset) return points[0].price;
  const last = points[points.length - 1];
  if (elapsed >= last.offset) return last.price;

  let low = 0;
  let high = points.length - 1;
  while (high - low > 1) {
    const mid = (low + high) >> 1;
    if (points[mid].offset <= elapsed) low = mid;
    else high = mid;
  }
  const a = points[low];
  const b = points[high];
  const span = b.offset - a.offset || 1;
  return a.price + ((b.price - a.price) * (elapsed - a.offset)) / span;
}

export function csvPoints(candles?: CsvCandle[]): DatasetPoint[] {
  if (!Array.isArray(candles) || candles.length === 0) return [];
  const parsed = candles
    .map((candle) => ({ offset: toEpochSeconds(candle.timestamp), price: Number(candle.close) }))
    .filter((point) => Number.isFinite(point.offset) && Number.isFinite(point.price) && point.price > 0)
    .sort((a, b) => a.offset - b.offset);
  if (parsed.length === 0) return [];
  const start = parsed[0].offset;
  return parsed.map((point) => ({ offset: point.offset - start, price: point.price }));
}

  if (points.length < 2) return null;
  return Math.max(1, points[points.length - 1].offset);
}

  }
  const date = new Date(timestamp);
  return Number.isNaN(date.getTime()) ? NaN : Math.round(date.getTime() / 1000);
}
